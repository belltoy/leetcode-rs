1//! # 8. 字符串转换整数 (atoi)
2//!
3//! 难度 中等
4//!
5//! 请你来实现一个 atoi 函数，使其能将字符串转换成整数。
6//!
7//! 首先，该函数会根据需要丢弃无用的开头空格字符，直到寻找到第一个非空格的字符为止。接下来的转化规则如下：
8//!
9//! - 如果第一个非空字符为正或者负号时，则将该符号与之后面尽可能多的连续数字字符组合起来，形成一个有符号整数。
10//! - 假如第一个非空字符是数字，则直接将其与之后连续的数字字符组合起来，形成一个整数。
11//! - 该字符串在有效的整数部分之后也可能会存在多余的字符，那么这些字符可以被忽略，它们对函数不应该造成影响。
12//!
13//! 注意：假如该字符串中的第一个非空格字符不是一个有效整数字符、字符串为空或字符串仅包含空白字符时，则你的函数不需要进行转换，即无法进行有效转换。
14//!
15//! 在任何情况下，若函数不能进行有效的转换时，请返回 0 。
16//!
17//! 提示：
18//!
19//! - 本题中的空白字符只包括空格字符 ' ' 。
20//! - 假设我们的环境只能存储 32 位大小的有符号整数，那么其数值范围为 [−2<sup>31</sup>,  2<sup>31</sup> − 1]。如果数值超过这个范围，请返回  INT_MAX (2<sup>31</sup> − 1) 或 INT_MIN (−2<sup>31</sup>) 。
21//!
22//!
23//! ## 示例 1:
24//!
25//! ```text
26//! 输入: "42"
27//! 输出: 42
28//! ```
29//!
30//! ## 示例 2:
31//!
32//! ```text
33//! 输入: "   -42"
34//! 输出: -42
35//! 解释: 第一个非空白字符为 '-', 它是一个负号。
36//!      我们尽可能将负号与后面所有连续出现的数字组合起来，最后得到 -42 。
37//! ```
38//!
39//! ## 示例 3:
40//!
41//! ```text
42//! 输入: "4193 with words"
43//! 输出: 4193
44//! 解释: 转换截止于数字 '3' ，因为它的下一个字符不为数字。
45//! ```
46//!
47//! ## 示例 4:
48//!
49//! ```text
50//! 输入: "words and 987"
51//! 输出: 0
52//! 解释: 第一个非空字符是 'w', 但它不是数字或正、负号。
53//!      因此无法执行有效的转换。
54//! ```
55//!
56//! ## 示例 5:
57//!
58//! ```text
59//! 输入: "-91283472332"
60//! 输出: -2147483648
61//! 解释: 数字 "-91283472332" 超过 32 位有符号整数范围。
62//!      因此返回 INT_MIN (−231) 。
63//! ```
64//!
65//! See [leetcode](https://leetcode-cn.com/problems/string-to-integer-atoi/)
66
67pub struct Solution;
68
69impl Solution {
70
71    /// FSM
72    pub fn my_atoi(s: String) -> i32 {
73        s.chars()
74         .try_fold(State::new(), State::handle)
75         .map_or_else(|n|n, State::into_ans)
76    }
77}
78
79enum State {
80    Start,
81    Sign(i32),
82    Num(i32),
83}
84
85impl State {
86    fn new() -> Self{
87        Self::Start
88    }
89
90    fn into_ans(self) -> i32 {
91        match self {
92            State::Num(n) => n,
93            _ => 0,
94        }
95    }
96
97    fn handle(self, c: char) -> Result<Self, i32> {
98        match self {
99            State::Start => match c {
100                ' ' => Ok(State::Start),
101                '+' => Ok(State::Sign(1)),
102                '-' => Ok(State::Sign(-1)),
103                c @ '0'..='9' => Ok(State::Num(c.to_digit(10).unwrap() as i32)),
104                _ => Err(0),
105            }
106            State::Sign(sign) => match c {
107                '0' => Ok(State::Sign(sign)),
108                c @ '1'..='9' => Ok(State::Num(sign * c.to_digit(10).unwrap() as i32)),
109                _ => Err(0),
110            }
111            State::Num(n) => match c {
112                c @ '0'..='9' => {
113                    if n >= 0 {
114                        n.checked_mul(10)
115                         .and_then(|n| n.checked_add(c.to_digit(10).unwrap() as i32))
116                         .map(|n| State::Num(n))
117                         .ok_or(i32::MAX)
118                    } else {
119                        n.checked_mul(10)
120                         .and_then(|n| n.checked_sub(c.to_digit(10).unwrap() as i32))
121                         .map(|n| State::Num(n))
122                         .ok_or(i32::MIN)
123                    }
124                }
125                _ => Err(n),
126            }
127        }
128    }
129}
