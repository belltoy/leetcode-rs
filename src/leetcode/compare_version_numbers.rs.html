1//! # 165. 比较版本号
2//!
3//! 难度 中等
4//!
5//! 给你两个版本号 `version1` 和 `version2` ，请你比较它们。
6//!
7//! 版本号由一个或多个修订号组成，各修订号由一个 `'.'` 连接。
8//! 每个修订号由 **多位数字** 组成，可能包含 **前导零** 。
9//! 每个版本号至少包含一个字符。修订号从左到右编号，下标从 0 开始，最左边的修订号下标为 0 ，下一个修订号下标为 1 ，以此类推。
10//! 例如，`2.5.33` 和 `0.1` 都是有效的版本号。
11//!
12//! 比较版本号时，请按从左到右的顺序依次比较它们的修订号。比较修订号时，只需比较 **忽略任何前导零后的整数值** 。
13//! 也就是说，修订号 `1` 和修订号 `001` 相等 。
14//! 如果版本号没有指定某个下标处的修订号，则该修订号视为 `0` 。
15//! 例如，版本 `1.0` 小于版本 `1.1` ，因为它们下标为 `0` 的修订号相同，而下标为 `1` 的修订号分别为 `0` 和 `1` ，`0 < 1` 。
16//!
17//! 返回规则如下：
18//!
19//! 如果 `version1 > version2` 返回 `1`，
20//! 如果 `version1 < version2` 返回 `-1`，
21//! 除此之外返回 `0`。
22//!
23//!
24//! ## 示例 1：
25//!
26//! ```text
27//! 输入：version1 = "1.01", version2 = "1.001"
28//! 输出：0
29//! 解释：忽略前导零，"01" 和 "001" 都表示相同的整数 "1"
30//! ```
31//!
32//! ## 示例 2：
33//!
34//! ```text
35//! 输入：version1 = "1.0", version2 = "1.0.0"
36//! 输出：0
37//! 解释：version1 没有指定下标为 2 的修订号，即视为 "0"
38//! ```
39//!
40//! ## 示例 3：
41//!
42//! ```text
43//! 输入：version1 = "0.1", version2 = "1.1"
44//! 输出：-1
45//! 解释：version1 中下标为 0 的修订号是 "0"，version2 中下标为 0 的修订号是 "1" 。0 < 1，所以 version1 < version2
46//! ```
47//!
48//! ## 示例 4：
49//!
50//! ```text
51//! 输入：version1 = "1.0.1", version2 = "1"
52//! 输出：1
53//! ```
54//!
55//! ## 示例 5：
56//!
57//! ```text
58//! 输入：version1 = "7.5.2.4", version2 = "7.5.3"
59//! 输出：-1
60//! ```
61//!
62//!
63//! ## 提示：
64//!
65//! - `1 <= version1.length, version2.length <= 500`
66//! - `version1` 和 `version2` 仅包含数字和 `'.'`
67//! - `version1` 和 `version2` 都是 **有效版本号**
68//! - `version1` 和 `version2` 的所有修订号都可以存储在 **32 位整数** 中
69//!
70//! See [leetcode](https://leetcode-cn.com/problems/compare-version-numbers/)
71
72pub struct Solution;
73
74impl Solution {
75    /// 使用 [`Iterator`](Iterator) 和 Pattern Matching 很简洁
76    ///
77    /// `s.split(".")` 拆成一个 Iterator，这种东西是 lazy 的，不需要调用 `collect()` 收集成 `Vec`，可以直接用 Iterator 的方式处理。
78    ///
79    /// 这里用了 `fuse()` 为了让 Iterator 可以持续调用 `next()` 直到两个 Iterator 都返回 `None` 才退出（说明比较完了）。
80    ///
81    /// 然后是 Pattern Matching 的方式，直接把所有情况列出。
82    ///
83    ///
84    /// - 相同的情况，继续比较下一个
85    /// - 不同的情况，可以直接返回结果
86    /// - 其中一个为 `None` 的情况，说明这个字符串已经结束了，相当于后面都是 `0`，只要另一个还有非零的值就比它大（这里题目假设了输入都是有效的，所以不可能是负数）。
87    /// - 最后全部比较完还没有差异，就是相同了，返回 `0`
88    ///
89    pub fn compare_version(version1: String, version2: String) -> i32 {
90        // we can use `parse()` and `unwrap()` because the inputs are assumed valid
91        let mut s1 = version1.split(".").fuse().map(|s| s.parse().unwrap());
92        let mut s2 = version2.split(".").fuse().map(|s| s.parse().unwrap());
93        loop {
94            match (s1.next(), s2.next()) {
95                (None, None) => break,
96                (Some(0), None) | (None, Some(0)) => continue,
97                (Some(_n1), None) => return 1,
98                (None, Some(_n2)) => return -1,
99                (Some(n1), Some(n2)) if n1 > n2 => return 1,
100                (Some(n1), Some(n2)) if n1 < n2 => return -1,
101                (Some(_n1), Some(_n2)) /*if n1 == n2*/ => continue,
102            }
103        }
104        0
105    }
106}
